standardize_compera <- function(x){
  x <- tolower(trimws(as.character(x)))
  dplyr::case_when(
    x == "low" ~ "low",
    x %in% c("intermediate-low", "intermediate low", "intermediate_low") ~ "intermediate-low",
    x %in% c("intermediate-high", "intermediate high", "intermediate_high") ~ "intermediate-high",
    x == "high" ~ "high",
    TRUE ~ NA_character_
  )
}

# ROC helpers --------------------------------------------------------
roc_curve_df <- function(y, score){
  ok <- is.finite(y) & is.finite(score)
  y <- y[ok]
  score <- score[ok]

  rr <- pROC::roc(
    y, score,
    levels = c(0, 1),
    direction = "<",
    quiet = TRUE
  )

  data.frame(
    FPR = 1 - rr$specificities,
    Sensitivity = rr$sensitivities
  ) %>%
    arrange(FPR, Sensitivity)
}

roc_bootstrap_band <- function(y, score, B = 2000, seed = 1,
                               grid = seq(0, 1, length.out = 101)){
  ok <- is.finite(y) & is.finite(score)
  y <- y[ok]
  score <- score[ok]

  set.seed(seed)
  sens_mat <- matrix(NA_real_, nrow = B, ncol = length(grid))

  for (b in seq_len(B)){
    idx <- sample(seq_along(y), replace = TRUE)
    yb <- y[idx]
    sb <- score[idx]

    if (length(unique(yb)) < 2) next

    dd <- roc_curve_df(yb, sb) %>%
      group_by(FPR) %>%
      summarise(Sensitivity = max(Sensitivity), .groups = "drop")

    dd <- bind_rows(
      data.frame(FPR = 0, Sensitivity = 0),
      dd,
      data.frame(FPR = 1, Sensitivity = 1)
    ) %>%
      group_by(FPR) %>%
      summarise(Sensitivity = max(Sensitivity), .groups = "drop") %>%
      arrange(FPR)

    sens_mat[b, ] <- approx(
      x = dd$FPR,
      y = dd$Sensitivity,
      xout = grid,
      ties = max,
      rule = 2
    )$y
  }

  data.frame(
    FPR = grid,
    lower = apply(sens_mat, 2, quantile, 0.025, na.rm = TRUE),
    median = apply(sens_mat, 2, quantile, 0.50, na.rm = TRUE),
    upper = apply(sens_mat, 2, quantile, 0.975, na.rm = TRUE)
  )
}

fmt_cont <- function(x, digits = 1){
  x <- as.numeric(x)
  x <- x[is.finite(x)]
  if (length(x) == 0) return("-")
  paste0(formatC(median(x), digits = digits, format = "f"), " [",
         formatC(quantile(x, 0.25), digits = digits, format = "f"), "–",
         formatC(quantile(x, 0.75), digits = digits, format = "f"), "], n=", length(x))
}

fmt_binary <- function(x, positive = "yes"){
  x <- tolower(trimws(as.character(x)))
  x <- x[!is.na(x) & x != ""]
  if (length(x) == 0) return("-")
  npos <- sum(x %in% positive)
  paste0(npos, " (", round(100 * npos / length(x), 1), "%), n=", length(x))
}

fmt_female <- function(x){
  x <- toupper(trimws(as.character(x)))
  x <- x[!is.na(x) & x != ""]
  if (length(x) == 0) return("-")
  npos <- sum(x == "F")
  paste0(npos, " (", round(100 * npos / length(x), 1), "%), n=", length(x))
}

fmt_compera <- function(x, level){
  x <- as.character(x)
  x <- x[!is.na(x)]
  if (length(x) == 0) return("-")
  nlevel <- sum(x == level)
  paste0(nlevel, " (", round(100 * nlevel / length(x), 1), "%), n=", length(x))
}

make_coupling_volcano <- function(res, title){
  keyvals <- rep("grey75", nrow(res))
  names(keyvals) <- rep("Not significant", nrow(res))
  keyvals[res$adj.P.Val < 0.1 & res$logFC > 0] <- "#B23A2B"
  keyvals[res$adj.P.Val < 0.1 & res$logFC < 0] <- "#2C6E9B"
  names(keyvals)[keyvals == "#B23A2B"] <- "Higher with impaired coupling"
  names(keyvals)[keyvals == "#2C6E9B"] <- "Lower with impaired coupling"

  EnhancedVolcano(
    res,
    lab = res$Protein,
    x = "logFC",
    y = "adj.P.Val",
    labSize = 0,
    pointSize = 1.5,
    pCutoff = 0.1,
    FCcutoff = 0,
    colCustom = keyvals,
    colAlpha = 1,
    subtitle = NULL,
    title = title,
    xlab = "NPX difference (impaired - preserved)"

Helper functions

make_stratified_folds <- function(y, k = 5){
  fold_id <- integer(length(y))
  for (cl in sort(unique(y))){
    ids <- sample(which(y == cl))
    fold_id[ids] <- rep(seq_len(k), length.out = length(ids))
  }
  fold_id
}
rank_proteins_limma <- function(expr_mat, meta, proteins){
  expr <- as.matrix(expr_mat[proteins, meta$Barcode, drop = FALSE])
  meta$Sex_M <- ifelse(as.character(meta$Sex) == "M", 1, 0)
  design <- model.matrix(~ Impaired + Age + Sex_M, data = meta)
  fit <- limma::eBayes(limma::lmFit(expr, design))
  limma::topTable(fit, coef = "Impaired", number = Inf, sort.by = "none") %>%
    rownames_to_column("Protein") %>% arrange(desc(abs(t)))
}
auc_safe <- function(y, pred){
  ok <- is.finite(y) & is.finite(pred)
  if (sum(ok) < 3 || length(unique(y[ok])) < 2) return(NA_real_)
  as.numeric(pROC::auc(pROC::roc(y[ok], pred[ok], levels = c(0, 1), direction = "<", quiet = TRUE)))
}

LASSO panel size: 1-10 proteins

For every outer training fold, candidate ranking is repeated from scratch, the top 50 proteins are passed to LASSO, and an inner cross-validation identifies the best solution for each panel size from 1 to 10 proteins. Performance across panel sizes is shown for transparency. For the final analysis, the panel size is fixed at five proteins.

choose_lasso_lambda_for_k <- function(cvfit, k, kmax = 10){
  beta <- as.matrix(coef(cvfit$glmnet.fit))
  nzero <- colSums(beta[-1, , drop = FALSE] != 0)
  idx <- which(nzero == k & is.finite(cvfit$cvm))

  if (length(idx) == 0){
    allowed <- which(nzero >= 1 & nzero <= kmax & is.finite(cvfit$cvm))
    if (length(allowed) == 0) return(NULL)
    distance <- abs(nzero[allowed] - k)
    allowed <- allowed[distance == min(distance)]
    idx <- allowed
  }

  best <- idx[which.max(cvfit$cvm[idx])]
  list(lambda = cvfit$lambda[best], nzero = nzero[best], cv_auc = cvfit$cvm[best])
}
run_lasso_nested_cv <- function(expr_mat, meta, proteins, candidate_n = 50, kmax = 10, repeats = 100, outer_folds = 5, inner_folds = 5, seed = 1){
  set.seed(seed)
  y <- meta$Impaired
  n <- nrow(meta)
  result_list <- vector("list", repeats * kmax)
  pred_sum <- matrix(0, nrow = n, ncol = kmax, dimnames = list(meta$Barcode, paste0("k", 1:kmax)))
  pred_n <- matrix(0, nrow = n, ncol = kmax, dimnames = list(meta$Barcode, paste0("k", 1:kmax)))
  counter <- 1

  for (r in seq_len(repeats)){
    outer_id <- make_stratified_folds(y, outer_folds)
    pred_repeat <- matrix(NA_real_, nrow = n, ncol = kmax)

    for (f in seq_len(outer_folds)){
      tr <- which(outer_id != f)
      te <- which(outer_id == f)
      meta_tr <- meta[tr, , drop = FALSE]
      meta_te <- meta[te, , drop = FALSE]

      ranking <- rank_proteins_limma(expr_mat, meta_tr, proteins)
      candidates <- ranking$Protein[seq_len(min(candidate_n, nrow(ranking)))]
      Xtr <- t(as.matrix(expr_mat[candidates, meta_tr$Barcode, drop = FALSE]))
      Xte <- t(as.matrix(expr_mat[candidates, meta_te$Barcode, drop = FALSE]))
      ytr <- meta_tr$Impaired

      inner_id <- make_stratified_folds(ytr, inner_folds)
      cvfit <- cv.glmnet(Xtr, ytr, family = "binomial", alpha = 1, type.measure = "auc", foldid = inner_id,
                         nlambda = 500, lambda.min.ratio = 1e-4, standardize = TRUE, dfmax = kmax)

      for (k in seq_len(kmax)){
        choice <- choose_lasso_lambda_for_k(cvfit, k, kmax)
        if (is.null(choice)) next
        pred_repeat[te, k] <- as.numeric(predict(cvfit$glmnet.fit, newx = Xte, s = choice$lambda, type = "response"))
      }
    }

    for (k in seq_len(kmax)){
      pred <- pred_repeat[, k]
      ok <- is.finite(pred)
      result_list[[counter]] <- data.frame(rep_id = r, k = k, AUC = auc_safe(y, pred))
      counter <- counter + 1
      pred_sum[ok, k] <- pred_sum[ok, k] + pred[ok]
      pred_n[ok, k] <- pred_n[ok, k] + 1
    }
  }

  results <- bind_rows(result_list)
  summary <- results %>% group_by(k) %>% summarise(
    mean_AUC = mean(AUC, na.rm = TRUE), sd_AUC = sd(AUC, na.rm = TRUE),
    se_AUC = sd_AUC / sqrt(sum(is.finite(AUC))),
    AUC_low = quantile(AUC, 0.025, na.rm = TRUE), AUC_high = quantile(AUC, 0.975, na.rm = TRUE), .groups = "drop")
  pred_avg <- pred_sum / pred_n
  list(results = results, summary = summary, pred_avg = pred_avg)
}

Select the final LASSO panel in Krakow

After the panel size is fixed, the top 50 candidates from the full Krakow development cohort are kept fixed. LASSO is fitted in the full Krakow development cohort, and the solution with the selected number of proteins and the highest internal cross-validated AUC is used as the final panel. Bootstrap selection frequencies are calculated afterwards as a stability analysis and do not determine the final proteins.

select_best_panel_for_k <- function(cvfit, k){
  beta <- as.matrix(coef(cvfit$glmnet.fit))
  protein_names <- rownames(beta)[-1]
  nzero <- colSums(beta[-1, , drop = FALSE] != 0)
  valid <- which(is.finite(cvfit$cvm))
  exact <- valid[nzero[valid] == k]

  if (length(exact) > 0){
    best <- exact[which.max(cvfit$cvm[exact])]
    co <- beta[-1, best]
    names(co) <- protein_names
    selected <- names(co)[co != 0]
  } else {
    above <- valid[nzero[valid] > k]
    if (length(above) == 0) stop("No LASSO solution with at least the selected number of proteins was found.")
    closest_n <- min(nzero[above])
    closest <- above[nzero[above] == closest_n]
    best <- closest[which.max(cvfit$cvm[closest])]
    co <- beta[-1, best]
    names(co) <- protein_names
    co <- co[co != 0]
    selected <- names(sort(abs(co), decreasing = TRUE))[seq_len(k)]
  }

  list(proteins = selected, lambda = cvfit$lambda[best], cv_auc = cvfit$cvm[best], path_n = nzero[best])
}

Bootstrap LASSO stability

The final panel is already fixed above. The same top 50 candidate proteins are kept fixed in the bootstrap analysis. Each bootstrap LASSO selects five proteins. Selection frequency is used only to assess stability and does not determine the final panel.

bootstrap_lasso_selection <- function(expr_mat, meta, candidates, k, B = 500, inner_folds = 5, seed = 1){
  set.seed(seed)
  selected_lists <- vector("list", B)

  for (b in seq_len(B)){
    ids <- sample(seq_len(nrow(meta)), nrow(meta), replace = TRUE)
    meta_b <- meta[ids, , drop = FALSE]
    y <- meta_b$Impaired
    if (length(unique(y)) < 2) next

    X <- t(as.matrix(expr_mat[candidates, meta_b$Barcode, drop = FALSE]))
    folds_here <- min(inner_folds, min(table(y)))
    if (folds_here < 3) next
    foldid <- make_stratified_folds(y, folds_here)

    cvfit <- try(
      cv.glmnet(
        X, y, family = "binomial", alpha = 1, type.measure = "auc",
        foldid = foldid, nlambda = 500, lambda.min.ratio = 1e-4,
        standardize = TRUE, dfmax = 10
      ),
      silent = TRUE
    )
    if (inherits(cvfit, "try-error")) next

    choice <- try(select_best_panel_for_k(cvfit, k), silent = TRUE)
    if (inherits(choice, "try-error") || is.null(choice)) next
    selected_lists[[b]] <- choice$proteins
  }

  selected_lists <- selected_lists[lengths(selected_lists) == k]
  if (length(selected_lists) == 0) stop("No valid bootstrap LASSO models were obtained.")

  selection_matrix <- matrix(
    0,
    nrow = length(selected_lists),
    ncol = length(candidates),
    dimnames = list(NULL, candidates)
  )

  for (i in seq_along(selected_lists)){
    selection_matrix[i, selected_lists[[i]]] <- 1
  }

  selection_frequency <- colMeans(selection_matrix)

  summary <- data.frame(
    Protein = candidates,
    Selection_frequency = selection_frequency,
    row.names = NULL
  ) %>%
    arrange(desc(Selection_frequency))

  list(
    summary = summary,
    selected_lists = selected_lists,
    selection_matrix = selection_matrix,
    n_valid = nrow(selection_matrix)
  )
}

A standard logistic model is fitted using the final selected proteins. Protein means and standard deviations, model coefficients, and score standardization are all estimated in Krakow and then applied unchanged to Zurich.

fit_final_panel <- function(expr_mat, clin_df, proteins){
  X <- t(as.matrix(expr_mat[proteins, clin_df$Barcode, drop = FALSE]))
  mu <- colMeans(X)
  sdv <- apply(X, 2, sd)
  sdv[sdv == 0 | is.na(sdv)] <- 1

  Z <- sweep(sweep(X, 2, mu, "-"), 2, sdv, "/")
  y <- clin_df$y_impaired

  fit <- suppressWarnings(
    glm.fit(cbind(Intercept = 1, Z), y, family = binomial())
  )

  beta <- fit$coefficients
  names(beta) <- c("Intercept", colnames(Z))

  if (!fit$converged){
    warning("Final logistic model did not fully converge. Inspect the final panel before interpretation.")
  }

  if (any(!is.finite(beta))){
    stop("Non-finite coefficient in final logistic model. Inspect separation in the selected panel.")
  }

  raw_score <- as.numeric(cbind(Intercept = 1, Z) %*% beta)
  score_center <- mean(raw_score)
  score_scale <- sd(raw_score)
  if (score_scale == 0 || is.na(score_scale)) score_scale <- 1

  list(
    proteins = proteins,
    mu = mu,
    sd = sdv,
    beta = beta,
    score_center = score_center,
    score_scale = score_scale
  )
}

predict_final_panel <- function(model, expr_mat, barcodes){
  X <- t(as.matrix(expr_mat[model$proteins, barcodes, drop = FALSE]))
  Z <- sweep(sweep(X, 2, model$mu, "-"), 2, model$sd, "/")

  raw_score <- as.numeric(
    cbind(Intercept = 1, Z) %*% model$beta
  )

  data.frame(
    Barcode = barcodes,
    probability_impaired = plogis(raw_score),
    panel_score = (raw_score - model$score_center) / model$score_scale
  )
}

fit_score_linear <- function(data, formula, cohort_name, model_name){
  vars <- all.vars(formula)
  dd <- data %>%
    filter(Cohort == cohort_name) %>%
    filter(if_all(all_of(vars), ~ !is.na(.)))

  if (nrow(dd) <= length(vars)){
    return(data.frame(
      Cohort = cohort_name,
      Model = model_name,
      n = nrow(dd),
      Beta = NA_real_,
      CI_low = NA_real_,
      CI_high = NA_real_,
      p = NA_real_,
      Adjusted_R2 = NA_real_
    ))
  }

  fit <- lm(formula, data = dd)
  tt <- broom::tidy(fit, conf.int = TRUE) %>%
    filter(term == "panel_score")

  data.frame(
    Cohort = cohort_name,
    Model = model_name,
    n = nrow(dd),
    Beta = tt$estimate,
    CI_low = tt$conf.low,
    CI_high = tt$conf.high,
    p = tt$p.value,
    Adjusted_R2 = summary(fit)$adj.r.squared
  )
}

No differential-abundance preselection is used for this analysis. Proteomics models use all proteins measured in both cohorts. Alpha and lambda are tuned in the inner cross-validation. For the combined model, the DETECT score is forced into the model and only protein coefficients are penalized.

stratified_folds_binary <- function(y, k = 5){
  folds <- integer(length(y))
  for (lev in sort(unique(y))){
    idx <- sample(which(y == lev))
    folds[idx] <- rep(seq_len(k), length.out = length(idx))
  }
  folds
}

specificity_at_sensitivity <- function(y, score, target_sensitivity = 0.95){
  ok <- is.finite(y) & is.finite(score)
  y <- y[ok]
  score <- score[ok]

  thresholds <- sort(unique(c(Inf, score, -Inf)), decreasing = TRUE)
  out <- lapply(thresholds, function(th){
    pred <- ifelse(score >= th, 1, 0)
    sens <- sum(pred == 1 & y == 1) / sum(y == 1)
    spec <- sum(pred == 0 & y == 0) / sum(y == 0)
    c(threshold = th, sensitivity = sens, specificity = spec)
  }) %>% do.call(rbind, .) %>% as.data.frame()

  out <- out[out$sensitivity >= target_sensitivity, , drop = FALSE]
  if (nrow(out) == 0) return(c(threshold = NA, sensitivity = NA, specificity = NA))

  best <- out[out$specificity == max(out$specificity), , drop = FALSE]
  best <- best[which.max(best$threshold), , drop = FALSE]
  unlist(best[1, ])
}

calc_auc <- function(y, score){
  as.numeric(pROC::auc(

fit_elastic_net_inner <- function(X, y, foldid, alphas = c(0, 0.25, 0.5, 0.75, 1), penalty.factor = NULL){
  if (is.null(penalty.factor)){
    penalty.factor <- rep(1, ncol(X))
  }

  fits <- vector("list", length(alphas))
  cv_auc <- rep(NA_real_, length(alphas))

  for (i in seq_along(alphas)){
    fits[[i]] <- try(
      cv.glmnet(
        X, y,
        family = "binomial",
        alpha = alphas[i],
        foldid = foldid,
        type.measure = "auc",
        standardize = TRUE,
        nlambda = 100,
        penalty.factor = penalty.factor
      ),
      silent = TRUE
    )

    if (!inherits(fits[[i]], "try-error") && any(is.finite(fits[[i]]$cvm))){
      cv_auc[i] <- max(fits[[i]]$cvm, na.rm = TRUE)
    }
  }

  if (all(!is.finite(cv_auc))) return(NULL)

  best_i <- which.max(cv_auc)
  best_fit <- fits[[best_i]]
  best_lambda_i <- which.max(best_fit$cvm)

  list(
    fit = best_fit,
    alpha = alphas[best_i],
    lambda = best_fit$lambda[best_lambda_i],
    cv_auc = best_fit$cvm[best_lambda_i]
  )
}

for (r in seq_len(R_detect)){
  outer_folds <- stratified_folds_binary(y_detect, K_detect)

  for (f in seq_len(K_detect)){
    tr <- which(outer_folds != f)
    te <- which(outer_folds == f)

    p_detect <- detect_score[te]

    inner_k <- min(5, min(table(y_detect[tr])))
    if (inner_k < 3) next
    inner_fold <- stratified_folds_binary(y_detect[tr], inner_k)

    prot_fit <- fit_elastic_net_inner(
      X_detect_prot[tr, , drop = FALSE],
      y_detect[tr],
      foldid = inner_fold,
      alphas = alpha_grid
    )

    X_comb_tr <- cbind(DETECT = detect_score[tr], X_detect_prot[tr, , drop = FALSE])
    X_comb_te <- cbind(DETECT = detect_score[te], X_detect_prot[te, , drop = FALSE])
    penalty_comb <- c(0, rep(1, ncol(X_detect_prot)))

    comb_fit <- fit_elastic_net_inner(
      X_comb_tr,
      y_detect[tr],
      foldid = inner_fold,
      alphas = alpha_grid,
      penalty.factor = penalty_comb
    )

    if (is.null(prot_fit) || is.null(comb_fit)) next

    p_prot <- as.numeric(
      predict(prot_fit$fit$glmnet.fit, newx = X_detect_prot[te, , drop = FALSE],
              s = prot_fit$lambda, type = "response")
    )

    p_comb <- as.numeric(
      predict(comb_fit$fit$glmnet.fit, newx = X_comb_te,
              s = comb_fit$lambda, type = "response")
    )

    pred_sum_detect[te, "DETECT"] <- pred_sum_detect[te, "DETECT"] + p_detect
    pred_sum_detect[te, "Proteomics_elastic_net"] <- pred_sum_detect[te, "Proteomics_elastic_net"] + p_prot
    pred_sum_detect[te, "DETECT_plus_proteomics_elastic_net"] <- pred_sum_detect[te, "DETECT_plus_proteomics_elastic_net"] + p_comb
    pred_n_detect[te, ] <- pred_n_detect[te, ] + 1

    beta_prot <- as.matrix(coef(prot_fit$fit$glmnet.fit, s = prot_fit$lambda))[-1, 1]
    beta_comb <- as.matrix(coef(comb_fit$fit$glmnet.fit, s = comb_fit$lambda))[-1, 1]
    beta_comb_proteins <- beta_comb[-1]

    detect_tuning_list[[counter_detect]] <- data.frame(
      rep_id = r,
      fold = f,
      proteomics_alpha = prot_fit$alpha,
      proteomics_inner_AUC = prot_fit$cv_auc,
      proteomics_nonzero = sum(beta_prot != 0),
      combined_alpha = comb_fit$alpha,
      combined_inner_AUC = comb_fit$cv_auc,
      combined_nonzero_proteins = sum(beta_comb_proteins != 0)
    )
    counter_detect <- counter_detect + 1
  }
